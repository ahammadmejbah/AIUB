//complex view
CREATE VIEW dept_sum_vu (name, minsal, maxsal, avgsal) AS 
SELECT	d.dname, MIN(e.sal), MAX(e.sal), ROUND(AVG(e.sal))
FROM emp e, dept d
WHERE e.deptno = d.deptno
GROUP BY d.dname;

select * from dept_sum_vu
