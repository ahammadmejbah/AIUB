CREATE TABLE STUDENT (
    SID NUMBER(3) NOT NULL,
    SNAME VARCHAR2(20) NOT NULL,
    Age INT,
    CHECK (Age>=18),
    EMAIL VARCHAR2(20),
    UNIQUE (EMAIL),
    SCGPA DECIMAL(3,2) DEFAULT 0.00,
    ADMISSION DATE,
    DUE_BALANCE FLOAT(8),
    CONSTRAINT PK_STUDENT PRIMARY KEY (SID)
);
