**** CREATE A SIMPLE VIEW NAMELY EMPVIEW *********
CREATE VIEW EMPVIEW10 AS SELECT EMPNO, JOB, ENAME FROM EMP WHERE DEPTNO IN (10,20);

***DISPLAY VIEW****
SELECT * FROM EMPVIEW10;

*****COLUMN ALIASES*****
CREATE VIEW EMPVIEW30 AS SELECT EMPNO "EMPLOYEEID", ENAME "EMPLOYEENAME", SAL "MONTHLYSALARY", SAL*12 "ANNUALSALARY", JOB "EMPLOYEEJOB" 
FROM EMP WHERE DEPTNO=20; 

******QUERY IN VIEW*********
SELECT MAX(ANNUALSALARY), MIN(ANNUALSALARY), EMPLOYEEJOB FROM EMPVIEW30 GROUP BY EMPLOYEEJOB;

******COMPLEX VIEW**********
CREATE VIEW EMP_DEPT (DEPARTMENT_NAME,MINIMUM_SALARY,AVERAGE_SALARY,MAXIMUM_SALARY) AS SELECT DNAME, MIN(SAL), AVG(SAL), MAX(SAL)
FROM EMP,DEPT WHERE EMP.DEPTNO=DEPT.DEPTNO GROUP BY DNAME;

******DISPLAY VIEW***********
SELECT * FROM EMP_DEPT;

*****BEFORE MODIFICATION********
SELECT * FROM EMPVIEW10;

*****MODIFY VIEW*********
CREATE OR REPLACE VIEW EMPVIEW10(EMPLOYEE_NAME,EMPLOYEE_ID,EMPLOYEE_JOB,SALARY,DEPARTMENT) AS SELECT ENAME, EMPNO, JOB, SAL, DEPTNO 
FROM EMP WHERE DEPTNO=30;

****AFTER MODIFICATION*****
SELECT * FROM EMPVIEW10;

*****DML ON VIEW*****
DELETE FROM EMPVIEW10 WHERE EMPLOYEE_NAME='JAMES';

*****READ ONLY VIEW*******
CREATE VIEW EMPVIEW55 (NAME,SALARY) AS SELECT ENAME, SAL FROM EMP WITH READ ONLY; 

*****DML ON READ ONLY VIEW********
DELETE FROM EMPVIEW55 WHERE NAME='KING';

****DISTROY VIEW**** 
DROP VIEW EMPVIEW10;
