CREATE TABLE Bus_Tickets (
    Ticket_No VARCHAR(10) PRIMARY KEY,
    Passenger_ID VARCHAR(10),
    Travel_Date DATE,
    Seat_No INT,
    Status VARCHAR(20),
    Bus_No VARCHAR(10),
    Route_No VARCHAR(10)
);


CREATE TABLE Buses (
    Bus_No VARCHAR(10) PRIMARY KEY,
    Seating_Capacity INT,
    Type VARCHAR(10),
    Driver_Name VARCHAR(100)
);


CREATE TABLE Route (
    Route_No VARCHAR(10) PRIMARY KEY,
    Start_Point VARCHAR(100),
    Destination VARCHAR(100),
    Total_Distance VARCHAR(20)
);


CREATE TABLE Passenger (
    Passenger_ID VARCHAR(10) PRIMARY KEY,
    Passenger_Name VARCHAR(100),
    Contact_No VARCHAR(15)
);

-- Simple View
CREATE VIEW Simple_View AS
SELECT 
    BT.Ticket_No,P.Passenger_Name,BT.Travel_Date
FROM 
    Bus_Tickets BT
JOIN 
    Passenger P ON BT.Passenger_ID = P.Passenger_ID;


select * from Simple_View;
DESCRIBE Simple_View;



-- Complex view 

CREATE VIEW Complex_View AS
SELECT 
    BT.Ticket_No,P.Passenger_Name,BT.Travel_Date,BT.Seat_No,B.Bus_No,B.Driver_Name,R.Start_Point,R.Destination,R.Total_Distance
FROM 
    Bus_Tickets BT
INNER JOIN 
    Passenger P ON BT.Passenger_ID = P.Passenger_ID
LEFT JOIN 
    Buses B ON BT.Bus_No = B.Bus_No
FULL OUTER JOIN 
    Route R ON BT.Route_No = R.Route_No
WHERE 
    BT.Status = 'Confirmed' OR BT.Status IS NULL;

DESCRIBE Complex_View;
